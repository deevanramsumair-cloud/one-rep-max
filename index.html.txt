  <script>
    let unit = 'kg';

    function setUnit(newUnit) {
      const weightInput = document.getElementById('weight');
      const currentVal  = parseFloat(weightInput.value);

      // Convert the displayed value before switching
      if (!isNaN(currentVal) && currentVal > 0) {
        if (newUnit === 'lbs' && unit === 'kg') {
          weightInput.value = (currentVal * 2.20462).toFixed(1);
        } else if (newUnit === 'kg' && unit === 'lbs') {
          weightInput.value = (currentVal / 2.20462).toFixed(1);
        }
      }

      unit = newUnit;
      document.getElementById('weight-unit-tag').textContent = newUnit;
      document.getElementById('btn-kg').classList.toggle('active',  newUnit === 'kg');
      document.getElementById('btn-lbs').classList.toggle('active', newUnit === 'lbs');

      // Recalculate with new unit if results are already showing
      if (document.getElementById('results').classList.contains('visible')) {
        calculate();
      }
    }

    function calculate() {
      const w   = parseFloat(document.getElementById('weight').value);
      const r   = parseInt(document.getElementById('reps').value);
      const err = document.getElementById('error');
      const res = document.getElementById('results');

      if (!w || !r || w <= 0 || r < 1 || isNaN(w) || isNaN(r)) {
        err.classList.add('visible');
        res.classList.remove('visible');
        return;
      }
      err.classList.remove('visible');

      const orm = w * (1 + r / 30);

      document.getElementById('out-weight').textContent  = w.toFixed(1) + ' ' + unit;
      document.getElementById('out-reps').textContent    = r + ' reps';
      document.getElementById('out-formula').textContent = w.toFixed(1) + ' × (1 + ' + r + '/30)';
      document.getElementById('out-total').textContent   = orm.toFixed(1) + ' ' + unit;

res.classList.add('visible');
    }

    function reset() {
      document.getElementById('weight').value = '';
      document.getElementById('reps').value   = '';
      document.getElementById('error').classList.remove('visible');
      document.getElementById('results').classList.remove('visible');
      document.getElementById('weight').focus();
    }

    function copyResult() {
      const total = document.getElementById('out-total').textContent;
      if (total === '—') return;
      const w = document.getElementById('weight').value;
      const r = document.getElementById('reps').value;
      const text = `1RM estimate: ${total} — based on ${w} ${unit} × ${r} reps (Epley formula)`;
      navigator.clipboard.writeText(text).then(() => {
        const btn = document.getElementById('copy-btn');
        btn.textContent = '✓ Copied!';
        setTimeout(() => { btn.textContent = '⎘ Copy result'; }, 2000);
      });
    }

    document.addEventListener('keydown', e => { if (e.key === 'Enter') calculate(); });
  </script>
